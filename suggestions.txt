logger = logging.getLogger(__name__)

class FlowFilter:
    def __init__(self, ticker: str = "COIN", min_premium: float = 500000.0, max_dte: int = 30):
        self.ticker = ticker
        self.min_premium = min_premium
        self.max_dte = max_dte

        self.bias = "NEUTRAL"
        self.bias_expires_at = datetime.min
        self.lock = asyncio.Lock()
        self.queue = asyncio.Queue()
        self.is_connected = False

    async def process_queue_loop(self):
        """Worker loop processing trades out-of-band to prevent event loop lag."""
        while True:
            try:
                msg_text = await self.queue.get()
                trade = json.loads(msg_text)

                # Verify criteria: COIN, premium threshold, short-term DTE
                if trade.get("ticker") == self.ticker:
                    premium = float(trade.get("premium", 0))
                    dte = int(trade.get("dte", 999))

                    if premium >= self.min_premium and dte <= self.max_dte:
                        is_call = trade.get("is_call", True)
                        async with self.lock:
                            self.bias = "BULLISH" if is_call else "BEARISH"
                            self.bias_expires_at = datetime.utcnow() + timedelta(hours=2)
                            logger.info(f"FlowFilter updated: {self.bias} until {self.bias_expires_at}")

                self.queue.task_done()
            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.error(f"Error parsing raw flow packet: {e}")

    async def get_current_bias(self) -> str:
        """Thread-safe look at current directional filter."""
        async with self.lock:
            if datetime.utcnow() > self.bias_expires_at:
                self.bias = "NEUTRAL"
            return self.bias

    async def allows(self, action_side: str) -> bool:
        """Validates strategy signals against institutional momentum."""
        current_bias = await self.get_current_bias()
        if current_bias == "NEUTRAL":
            return True
        if current_bias == "BULLISH" and action_side.upper() == "LONG":
            return True
        if current_bias == "BEARISH" and action_side.upper() == "SHORT":
            return True
        return False

    async def handle_disconnect(self):
        """Safety reset preventing stale filter application during network drops."""
        async with self.lock:
            self.bias = "NEUTRAL"
            self.bias_expires_at = datetime.min
            self.is_connected = False
            logger.warning("Unusual Whales disconnected. Safety reset to NEUTRAL activated.")

class StrategyOrderManager:
    def __init__(self, trading_client: TradingClient):
        self.client = trading_client

    def submit_strategy_bracket(self, strategy_name: str, side: str, qty: int, entry_symbol: str, target_price: float, stop_price: float):
        """
        Executes isolated bracket orders using prefix tags inside client_order_id.
        Prevents multiple running strategies from overlapping or liquidating each other.
        """
        order_side = OrderSide.BUY if side.upper() == "LONG" else OrderSide.SELL
        opposite_side = OrderSide.SELL if order_side == OrderSide.BUY else OrderSide.BUY

        # Tag order uniqueness structurally
        unique_id = uuid.uuid4().hex[:8]
        tagged_client_id = f"{strategy_name}_{unique_id}"

        bracket_request = MarketOrderRequest(
            symbol=entry_symbol,
            qty=qty,
            side=order_side,
            time_in_force=TimeInForce.GTC,
            client_order_id=tagged_client_id,
            order_class="bracket",
            take_profit=TakeProfitRequest(limit_price=round(target_price, 2)),
            stop_loss=StopLossRequest(stop_price=round(stop_price, 2))
        )

        try:
            order = self.client.submit_order(order_request=bracket_request)
            logger.info(f"Bracket executed safely for {strategy_name}. Order ID: {order.id}")
            return order
        except Exception as e:
            logger.error(f"Critical execution failure for {strategy_name}: {e}")
            return None

from typing import Optional

logger = logging.getLogger(__name__)

class FirstCandleBreakStrategy:
    def __init__(self, config: dict, opening_bar_atr: Optional[float]):
        self.config = config["fcb"]
        self.opening_bar_atr = opening_bar_atr
        self.has_fired_today = False

    def evaluate_935_bar(self, first_candle: dict, current_price: float, current_equity: float) -> Optional[dict]:
        """
        Evaluates the 9:35 AM execution window for COIN capitulation shorting.
        """
        if self.has_fired_today:
            return None

        fc_open = first_candle["open"]
        fc_close = first_candle["close"]
        fc_high = first_candle["high"]
        fc_low = first_candle["low"]
        fc_range = fc_high - fc_low

        # 1. First candle must be RED (bearish dominance)
        if fc_close >= fc_open:
            return None

        # 2. Dynamic Volatility Range Filter Floor
        if self.opening_bar_atr is not None:
            min_fc_range = self.config["atr_min_mult"] * self.opening_bar_atr
            if fc_range < min_fc_range:
                logger.info(f"FCB skipped: Range ({fc_range:.2f}) below volatility floor ({min_fc_range:.2f})")
                return None
        else:
            logger.warning("ATR context is missing. Strategy disabling itself safely.")
            return None

        # 3. Capitulation Guard (Avoid shorting exhaustive vertical gap downs)
        capitulation_threshold = fc_open * self.config["max_range_pct"]
        if fc_range > capitulation_threshold:
            logger.info(f"FCB skipped: Range ({fc_range:.2f}) exceeds capitulation safety ceiling ({capitulation_threshold:.2f})")
            return None

        # 4. Strategy Breakout Execution Trigger
        if current_price < fc_low:
            self.has_fired_today = True

            # Position sizing calculated via hard equity capital constraints
            risk_amount = current_equity * 0.02
            stop_distance = fc_range * self.config["stop_mult"]
            qty = max(1, int(risk_amount / stop_distance))

            target = current_price - (fc_range * self.config["target_mult"])
            stop = current_price + stop_distance

            return {
                "side": "SHORT",
                "qty": qty,
                "target": target,
                "stop": stop
            }

        return None
